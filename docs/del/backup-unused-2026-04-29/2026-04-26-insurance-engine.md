# Insurance Engine Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rebuild the insurance policy recognition system with template versioning, auto-learning, multi-OCR cross-validation, and structure change detection.

**Architecture:** A 6-stage pipeline (OCR → Company ID → Policy Type → Template Match → Field Extract → Validate) backed by per-company/type/version template JSON files and rule Python files. Templates are auto-generated from sample PDFs via a learning engine. Scanned PDFs use multi-OCR cross-validation; text-layer PDFs skip cloud OCR entirely.

**Tech Stack:** Python 3, Flask, pdfplumber, pymysql/DBUtils, requests (Baidu/Tencent OCR APIs), concurrent.futures

```python
def resolve_alert(db, alert_id: int, action: str):
    """标记告警已处理"""
    conn = db.pool.connection()
    try:
        cursor = conn.cursor()
        cursor.execute("""
            UPDATE template_alert_records SET resolved=1, resolved_action=%s WHERE id=%s
        """, (action, alert_id))
        conn.commit()
    finally:
        conn.close()


def query_alerts(db, page=1, page_size=20, resolved=None, level=None) -> dict:
    """查询告警记录"""
    conn = db.pool.connection()
    try:
        cursor = conn.cursor(pymysql.cursors.DictCursor)
        conditions = []
        params = []
        if resolved is not None:
            conditions.append("resolved=%s")
            params.append(resolved)
        if level:
            conditions.append("alert_level=%s")
            params.append(level)
        where = "WHERE " + " AND ".join(conditions) if conditions else ""
        cursor.execute(f"SELECT COUNT(*) as total FROM template_alert_records {where}", params)
        total = cursor.fetchone()["total"]
        offset = (page - 1) * page_size
        cursor.execute(f"""
            SELECT * FROM template_alert_records {where}
            ORDER BY created_at DESC LIMIT %s OFFSET %s
        """, params + [page_size, offset])
        records = cursor.fetchall()
        return {"total": total, "records": records, "page": page, "page_size": page_size}
    finally:
        conn.close()
```

```python
from dataclasses import dataclass, field
from typing import Dict, List, Optional, Any


@dataclass
class CompanyResult:
    """保司识别结果"""
    company_id: str = ""           # 如 "picc"
    company_name: str = ""         # 全称
    company_short: str = ""        # 简称

@dataclass
class PolicyTypeResult:
    """险种识别结果"""
    type_code: str = "unknown"     # compulsory / commercial / accident / unknown
    type_name: str = "未知"        # 交强险 / 商业险 / 驾乘/意外险
    raw_match: str = ""            # 匹配到的原始文本


@dataclass
class MatchResult:
    """模板匹配结果"""
    template: Optional[Dict] = None   # 匹配到的模板 JSON
    template_id: str = ""             # 如 "picc/compulsory/v1"
    score: float = 0.0
    status: str = "no_template"       # matched / partial_match / no_match / no_template
    warning: str = ""


@dataclass
class Alert:
    """告警"""
    level: str = "info"           # info / warning / critical
    alert_type: str = ""          # missing_required_fields / low_match_score / ocr_critical_diff / ...
    message: str = ""
    template_id: str = ""


@dataclass
class ValidationResult:
    """结构校验结果"""
    is_valid: bool = True
    alerts: List[Alert] = field(default_factory=list)


@dataclass
class FieldDiff:
    """OCR 字段差异"""
    field_name: str = ""
    layer: str = ""               # common / type_specific / company_specific
    values: Dict[str, str] = field(default_factory=dict)  # {engine: value}

    resolved_value: str = ""
    resolution_method: str = ""   # vote / priority


@dataclass
class CompareReport:
    """OCR 交叉比对报告"""
    merged_fields: Dict[str, Dict] = field(default_factory=dict)
    cross_validated: bool = False
    diffs: List[FieldDiff] = field(default_factory=list)
    engine_count: int = 1
    source_engine: str = ""

    @property

    validation: ValidationResult = field(default_factory=ValidationResult)

    confidence: float = 0.0
```

```python
class CompanyIdentifier:
    """保司识别 — 4 层策略"""

    def identify(self, text: str) -> CompanyResult:
        """识别保险公司"""
        company_name = ""
        raw_match = ""

        # 策略1：「公司名称：」标签格式
        m = re.search(r'公司名称[：:\s]*(.+?)(?:\s+公司地址|$)', text)
        if m:
            company_name = m.group(1).strip()
            raw_match = m.group(0)

        # 策略2：全称直接匹配
        if not company_name:
            for base in COMPANY_BASES:
                pattern = re.escape(base) + r'[\S]*'
                m = re.search(pattern, text)
                if m:
                    company_name = m.group(0)
                    # 清理尾部噪声
                    company_name = re.sub(
                        r'(公司|营业部|服务部|业务部|支公司).*', r'\1', company_name)
                    raw_match = m.group(0)
                    break

        # 策略3：特征信号降级
        if not company_name:
            for pattern, name in COMPANY_FALLBACK_SIGNALS:
                if re.search(pattern, text):
                    company_name = name
                    raw_match = pattern
                    break

        # 策略4：保单号前缀推断
        if not company_name:
            company_name, raw_match = self._infer_from_policy_no(text)

        # 映射简称和 ID
        company_short = self._get_short_name(company_name)
        company_id = COMPANY_ID_MAP.get(company_short, "")

        return CompanyResult(
            company_id=company_id,
            company_name=company_name,
            company_short=company_short,
            raw_match=raw_match,
        )

    def _get_short_name(self, company_name: str) -> str:
        """全称 → 简称"""
        for keyword, short in COMPANY_SHORT_MAP.items():
            if keyword in company_name:
                return short
        return company_name

    def _infer_from_policy_no(self, text: str) -> tuple:
        """从保单号前缀推断保司"""
        numbers = re.findall(r'[A-Za-z0-9]{15,30}', text)
        prefix_map = {

            r'^606EA': "华泰财产保险有限公司",

            r'^PICC': "中国人民财产保险股份有限公司",
        }
        for num in numbers:
            for pattern, name in prefix_map.items():
                if re.match(pattern, num):
                    return name, num
```

```python
    r"平安车主出行险",

class TypeIdentifier:
    """险种识别"""

    def identify(self, text: str) -> PolicyTypeResult:
        """识别险种类型"""
        # 标准模式匹配
        for pattern in POLICY_TYPE_PATTERNS:
            m = re.search(pattern, text)
            if m:
                raw = m.group(0)
                type_code, type_name = self._get_type_code(raw)
                return PolicyTypeResult(
                    type_code=type_code, type_name=type_name, raw_match=raw)

                    type_code="accident", type_name="驾乘/意外险",
                    raw_match="POLICY SCHEDULE (accident)")

                return PolicyTypeResult(
                    type_code="commercial", type_name="商业险",
                    raw_match="POLICY SCHEDULE (commercial)")

        return PolicyTypeResult()

            "驾乘", "意外", "出行险", "出行无忧", "安心行", "E车无忧", "卓越全意保"
        ]):
            return ("accident", "驾乘/意外险")
        return ("unknown", policy_type)
```

```python
"""pdfplumber 文本层提取引擎"""

import io
import re
import time
import pdfplumber
from insurance.ocr.base import BaseOCREngine, OCRResult


def clean_text(text: str) -> str:
    """清理 OCR 文本：去除数字间空格、日期单位间空格"""
    for _ in range(3):
        text = re.sub(r'(\d)\s+(\d)', r'\1\2', text)
    text = re.sub(r'(\d)\s+([年月日时分秒])', r'\1\2', text)
    text = re.sub(r'([年月日时分秒])\s+(\d)', r'\1\2', text)
    return text


class PdfplumberEngine(BaseOCREngine):
    engine_name = "pdfplumber"
    engine_type = "text_layer"

    def extract(self, pdf_bytes: bytes, page: int = 1) -> OCRResult:
        start = time.time()
        try:
            pdf_file = io.BytesIO(pdf_bytes)
            with pdfplumber.open(pdf_file) as pdf:
                page_count = len(pdf.pages)
                page_idx = min(page - 1, page_count - 1)
                page_obj = pdf.pages[page_idx]
                text = page_obj.extract_text() or ""

                if not text.strip():
                    return OCRResult(
                        engine=self.engine_name,
                        success=False,
                        error="无文本层（扫描件）",
                        page_count=page_count,
                        cost_ms=int((time.time() - start) * 1000),
                    )

                text = clean_text(text)
                return OCRResult(
                    engine=self.engine_name,
                    success=True,
                    text=text,
                    confidence=1.0,  # 文本层 100% 准确
                    page_count=page_count,
                    has_text_layer=True,
                    cost_ms=int((time.time() - start) * 1000),
                )
        except Exception as e:
            return OCRResult(
                engine=self.engine_name,
                success=False,
                error=str(e),
                cost_ms=int((time.time() - start) * 1000),
            )
```

```python
"""百度云 OCR 引擎"""

import base64
import time
import requests
from insurance.ocr.base import BaseOCREngine, OCRResult


class BaiduOCREngine(BaseOCREngine):
    engine_name = "baidu"
    engine_type = "image"

    TOKEN_URL = "https://aip.baidubce.com/oauth/2.0/token"
    OCR_URL = "https://aip.baidubce.com/rest/2.0/ocr/v1/accurate_basic"

    def __init__(self, api_key: str, secret_key: str):
        self.api_key = api_key
        self.secret_key = secret_key
        self._access_token = None
        self._token_expire_time = 0

    def extract(self, pdf_bytes: bytes, page: int = 1) -> OCRResult:
        start = time.time()
        try:
            token = self._get_access_token()
            pdf_base64 = base64.b64encode(pdf_bytes).decode()
            resp = requests.post(
                self.OCR_URL,
                params={"access_token": token},
                data={
                    "pdf_file": pdf_base64,
                    "pdf_file_num": str(page),
                    "language_type": "CHN_ENG",
                    "detect_direction": "true",
                },
                headers={"Content-Type": "application/x-www-form-urlencoded"},
                timeout=30,
            )
            result = resp.json()
            if "words_result" not in result:
                return OCRResult(
                    engine=self.engine_name, success=False,
                    error=result.get("error_msg", "未知错误"),
                    cost_ms=int((time.time() - start) * 1000))

            lines = [w["words"] for w in result["words_result"]]
            text = "\n".join(lines)
            return OCRResult(
                engine=self.engine_name, success=True, text=text,
                confidence=0.9,  # 百度 OCR 默认置信度
                cost_ms=int((time.time() - start) * 1000))
        except Exception as e:
            return OCRResult(
                engine=self.engine_name, success=False, error=str(e),
                cost_ms=int((time.time() - start) * 1000))

    def _get_access_token(self) -> str:
        if self._access_token and time.time() < self._token_expire_time:
            return self._access_token
        resp = requests.post(self.TOKEN_URL, params={
            "grant_type": "client_credentials",
            "client_id": self.api_key,
            "client_secret": self.secret_key,
        }, timeout=10)
        data = resp.json()
        self._access_token = data["access_token"]
        self._token_expire_time = time.time() + data.get("expires_in", 2592000) - 3600
        return self._access_token
```

```python
"""腾讯云 OCR 引擎"""

import base64
import hashlib
import hmac
import json
import time
from datetime import datetime, timezone
import requests
from insurance.ocr.base import BaseOCREngine, OCRResult


class TencentOCREngine(BaseOCREngine):
    engine_name = "tencent"
    engine_type = "image"

    API_HOST = "ocr.tencentcloudapi.com"

    def __init__(self, secret_id: str, secret_key: str):
        self.secret_id = secret_id
        self.secret_key = secret_key

    def extract(self, pdf_bytes: bytes, page: int = 1) -> OCRResult:
        start = time.time()
        try:
            pdf_base64 = base64.b64encode(pdf_bytes).decode()
            # 使用通用文字识别（高精度版）
            params = {
                "ImageBase64": pdf_base64,
                "IsPdf": True,
                "PdfPageNumber": page,
            }
            result = self._call_api("GeneralAccurateOCR", params)

            if "Error" in result.get("Response", {}):
                error = result["Response"]["Error"]
                return OCRResult(
                    engine=self.engine_name, success=False,
                    error=f"{error['Code']}: {error['Message']}",
                    cost_ms=int((time.time() - start) * 1000))

            text_detections = result.get("Response", {}).get("TextDetections", [])
            lines = [item["DetectedText"] for item in text_detections]
            text = "\n".join(lines)

            # 计算平均置信度
            confidences = [item.get("Confidence", 0) for item in text_detections]
            avg_confidence = sum(confidences) / len(confidences) / 100 if confidences else 0

            return OCRResult(
                engine=self.engine_name, success=True, text=text,
                confidence=avg_confidence,
                cost_ms=int((time.time() - start) * 1000))
        except Exception as e:
            return OCRResult(
                engine=self.engine_name, success=False, error=str(e),
                cost_ms=int((time.time() - start) * 1000))

    def _call_api(self, action: str, params: dict) -> dict:
        """腾讯云 API v3 签名调用"""
        timestamp = int(time.time())
        date = datetime.fromtimestamp(timestamp, tz=timezone.utc).strftime("%Y-%m-%d")
        payload = json.dumps(params)

        # 构造签名
        canonical_request = (
            f"POST\n/\n\n"
            f"content-type:application/json\n"
            f"host:{self.API_HOST}\n"
            f"x-tc-action:{action.lower()}\n\n"
            f"content-type;host;x-tc-action\n"
            f"{hashlib.sha256(payload.encode()).hexdigest()}"
        )
        credential_scope = f"{date}/ocr/tc3_request"
        string_to_sign = (
            f"TC3-HMAC-SHA256\n{timestamp}\n{credential_scope}\n"
            f"{hashlib.sha256(canonical_request.encode()).hexdigest()}"
        )

        def _hmac_sha256(key, msg):
            return hmac.new(key, msg.encode(), hashlib.sha256).digest()

        secret_date = _hmac_sha256(f"TC3{self.secret_key}".encode(), date)
        secret_service = _hmac_sha256(secret_date, "ocr")
        secret_signing = _hmac_sha256(secret_service, "tc3_request")
        signature = hmac.new(secret_signing, string_to_sign.encode(), hashlib.sha256).hexdigest()

        auth = (
            f"TC3-HMAC-SHA256 Credential={self.secret_id}/{credential_scope}, "
            f"SignedHeaders=content-type;host;x-tc-action, Signature={signature}"
        )

        resp = requests.post(
            f"https://{self.API_HOST}",
            headers={
                "Content-Type": "application/json",
                "Host": self.API_HOST,
                "X-TC-Action": action,
                "X-TC-Version": "2018-11-19",
                "X-TC-Timestamp": str(timestamp),
                "Authorization": auth,
            },
            data=payload,
            timeout=30,
        )
        return resp.json()
```

```python
"""OCR 多引擎调度器"""

import logging
from concurrent.futures import ThreadPoolExecutor, as_completed
from typing import List
from insurance.ocr.base import BaseOCREngine, OCRResult
from insurance.ocr.pdfplumber_engine import PdfplumberEngine

logger = logging.getLogger(__name__)


class OCRDispatcher:
    """根据 PDF 类型调度 OCR 引擎：有文本层直接用，扫描件多引擎并行"""

    def __init__(self, engines: List[BaseOCREngine] = None):
        self.pdfplumber = PdfplumberEngine()
        self.cloud_engines = [e for e in (engines or []) if e.engine_type == "image"]

    def dispatch(self, pdf_bytes: bytes, page: int = 1) -> List[OCRResult]:
        """调度 OCR 引擎，返回结果列表"""
        # 第一步：pdfplumber 提取文本层
        text_layer = self.pdfplumber.extract(pdf_bytes, page)

        if text_layer.success and text_layer.text.strip():
            # 有文本层 → 直接用，最准，零成本
            logger.info("PDF 有文本层，直接使用 pdfplumber")
            return [text_layer]

        # 无文本层（扫描件）→ 并行调所有云 OCR
        if not self.cloud_engines:
            logger.warning("扫描件但无可用云 OCR 引擎")
            return [text_layer]  # 返回失败结果

        logger.info(f"扫描件，并行调用 {len(self.cloud_engines)} 个云 OCR 引擎")
        results = []
        with ThreadPoolExecutor(max_workers=len(self.cloud_engines)) as pool:
            futures = {
                pool.submit(e.extract, pdf_bytes, page): e
                for e in self.cloud_engines
            }
            for future in as_completed(futures):
                try:
                    result = future.result()
                    results.append(result)
                except Exception as e:
                    engine = futures[future]
                    logger.error(f"OCR 引擎 {engine.engine_name} 异常: {e}")
                    results.append(OCRResult(
                        engine=engine.engine_name, success=False, error=str(e)))

        # 过滤成功的结果
        success_results = [r for r in results if r.success]
        if not success_results:
            logger.warning("所有云 OCR 引擎都失败了")
            return results  # 返回所有结果（含错误信息）

        return success_results
```

```python
"""OCR 交叉比对引擎"""

import re
import logging
from typing import List, Dict, Callable
from insurance.ocr.base import OCRResult
from insurance.engine.schema import FieldDiff, CompareReport

logger = logging.getLogger(__name__)

# 关键字段 — 差异必须标记
CRITICAL_FIELDS = [
    "保单号", "被保险人", "投保人", "车牌号",
    "保费合计", "保险起期", "保险止期",
]


class OCRComparator:
    """对多个 OCR 结果进行逐字段比对"""

    def compare(self, ocr_results: List[OCRResult],
                extract_fn: Callable) -> CompareReport:
        """
        用同一套提取逻辑分别解析每个 OCR 的文本，逐字段对比

        Args:
            ocr_results: 多个 OCR 引擎的结果
            extract_fn: 字段提取函数，签名 (text) -> dict（三层字段）
        """
        # 每个 OCR 结果分别走字段提取
        extractions = {}
        for result in ocr_results:
            if result.success and result.text.strip():
                fields = extract_fn(result.text)
                extractions[result.engine] = fields

        if len(extractions) == 0:
            return CompareReport()

        if len(extractions) == 1:
            engine = list(extractions.keys())[0]
            return CompareReport(
                merged_fields=extractions[engine],
                source_engine=engine,
                cross_validated=False,
                engine_count=1,
            )

        # 逐字段对比
        diffs = []
        merged = {}

        for layer in ["common", "type_specific", "company_specific"]:
            merged[layer] = {}
            all_fields = set()
            for eng in extractions:
                all_fields.update(extractions[eng].get(layer, {}).keys())

            for field_name in all_fields:
                values = {}
                for eng in extractions:
                    val = extractions[eng].get(layer, {}).get(field_name)
                    if val:
                        values[eng] = str(val)

                if not values:
                    merged[layer][field_name] = ""
                elif len(set(values.values())) <= 1:
                    # 所有引擎结果一致
                    merged[layer][field_name] = list(values.values())[0]
                else:
                    # 有差异
                    resolved = self._vote(values, ocr_results)
                    diff = FieldDiff(
                        field_name=field_name,
                        layer=layer,
                        values=values,
                        is_critical=field_name in CRITICAL_FIELDS,
                        resolved_value=resolved,
                        resolution_method="vote",
                    )
                    diffs.append(diff)
                    merged[layer][field_name] = resolved

        return CompareReport(
            merged_fields=merged,
            cross_validated=True,
            diffs=diffs,
            engine_count=len(extractions),
        )

    def _vote(self, values: Dict[str, str], ocr_results: List[OCRResult]) -> str:
        """投票策略：多数一致 → 选多数；无多数 → 按引擎置信度选"""
        # 统计每个值出现的引擎
        val_engines = {}
        for eng, val in values.items():
            val_engines.setdefault(val, []).append(eng)

        # 多数一致
        majority = max(val_engines.items(), key=lambda x: len(x[1]))
        if len(majority[1]) > len(values) / 2:
            return majority[0]

        # 无多数 → 按置信度选
        confidence_map = {r.engine: r.confidence for r in ocr_results}
        best_engine = max(values.keys(), key=lambda e: confidence_map.get(e, 0))
        return values[best_engine]
```

```python
import re
from typing import Dict, Any, List, Optional


PERSON_BLACKLIST = {
    "地址", "手机号", "信息", "名称", "保单验真码", "保单验真码:",
    "行驶证车主", "行驶证", "为", "身份证", "证件",

    "须如实告知",

def merge_text(text: str) -> str:
    """合并中文字符间的空格（处理 pdfplumber 拆字问题）"""
    return re.sub(r'([\u4e00-\u9fa5])\s+([\u4e00-\u9fa5])', r'\1\2', text)


def is_valid_person(name: str) -> bool:
    """校验人名是否合法"""
    if not name or len(name) < 2 or len(name) > 20:
        return False
    if name in PERSON_BLACKLIST:
        return False
    if re.match(r'^[：:（(]', name):
        return False
    if re.search(r'保险|公司|有限|股份|投保|被保', name):
        return False
    if re.search(r'\d{6,}|[A-Za-z]{5,}', name):
        return False
    return True


def clean_person_name(name: str) -> str:
    """清理人名"""
    name = re.sub(r'^[：:\s]+', '', name)
    name = re.sub(r'[（(].*$', '', name)
    name = re.sub(r'车主.*$', '', name)
    name = re.sub(r'名称.*$', '', name)
    name = re.sub(r'证件.*$', '', name)
    name = re.sub(r'有敬异的.*$', '', name)  # 永诚特有噪声
    return name.strip()


class BaseRule:
    """基础规则 — 所有保司所有险种通用的字段提取逻辑

    子类通过 override 单个 extract_* 方法来定制特定保司/版本的提取逻辑。
    """

    company_id: str = ""
    policy_type: str = ""
    version: str = ""

    def extract_all(self, text: str) -> Dict[str, Any]:
        """主入口：提取三层字段

        Args:
            text: OCR 提取的原始文本

        Returns:
            {
                "common": {...},
                "type_specific": {...},
                "company_specific": {...},
                "insurance_items": [...],
            }
        """
        text_merged = merge_text(text)
        return {
            "common": self._extract_common(text, text_merged),
            "type_specific": self.extract_type_specific(text, text_merged),
            "company_specific": self.extract_company_specific(text, text_merged),
            "insurance_items": self.extract_items(text, text_merged),
        }

    def _extract_common(self, text: str, text_merged: str) -> Dict[str, str]:
        """提取所有公共字段"""
        fields = {}
        fields["保单号"] = self.extract_policy_no(text, text_merged)
        fields["投保确认码"] = self.extract_confirm_code(text, text_merged)
        fields["被保险人"] = self.extract_insured(text, text_merged)
        fields["投保人"] = self.extract_proposer(text, text_merged)

        # 车辆信息
        vehicle = self.extract_vehicle_info(text, text_merged)
        fields.update(vehicle)

        # 保费
        premium = self.extract_premium(text, text_merged)
        fields.update(premium)

        # 车船税
        fields["车船税"] = self.extract_tax(text, text_merged)

        # 保险期间
        period = self.extract_period(text, text_merged)
        fields.update(period)

        # 签单日期
        fields["签单日期"] = self.extract_sign_date(text, text_merged)

        # 其他
        fields["收费确认时间"] = self.extract_fee_confirm_time(text, text_merged)
        fields["销售渠道"] = self.extract_sales_channel(text, text_merged)
        fields["中介机构"] = self.extract_intermediary(text, text_merged)
        fields["争议解决方式"] = self.extract_dispute_resolution(text, text_merged)
        fields["制单人"] = self.extract_underwriter(text, text_merged)
        fields["经办人"] = self.extract_handler_person(text, text_merged)

        # 互填：被保险人和投保人
        if fields.get("被保险人") and not fields.get("投保人"):
            fields["投保人"] = fields["被保险人"]
        elif fields.get("投保人") and not fields.get("被保险人"):
            fields["被保险人"] = fields["投保人"]

        return fields

    def extract_policy_no(self, text: str, text_merged: str) -> str:

            r'保险凭证号[：:\s]\s*(\S+)',

            m = re.search(pattern, text)
            if m:
                return m.group(1).strip()
            m = re.search(pattern, text_merged)
            if m:
                return m.group(1).strip()

            if re.match(r'^[A-Za-z0-9]{15,30}$', line):

    def extract_period(self, text: str, text_merged: str) -> Dict[str, str]:
        """保险期间提取"""
        fields = {"保险期间": "", "保险起期": "", "保险止期": ""}

        # 标准格式
        m = re.search(
            r'保险期间[：:\s]*(\d{4})[年-](\d{1,2})[月-](\d{1,2})[日]?\s*\S{0,5}起\S{0,2}\s*'
            r'(\d{4})[年-](\d{1,2})[月-](\d{1,2})[日]?',
            text
        )
        if m:
            start = f"{m.group(1)}年{m.group(2)}月{m.group(3)}日"
            end = f"{m.group(4)}年{m.group(5)}月{m.group(6)}日"
            fields["保险起期"] = start
            fields["保险止期"] = end
            fields["保险期间"] = f"{start}至{end}"
            return fields

        # 自...至...格式
        m = re.search(
            r'自[（(From)）]*\s*(\d{4})[年/-](\d{1,2})[月/-](\d{1,2})[日]?.*?'
            r'至[（(To)）]*\s*(\d{4})[年/-](\d{1,2})[月/-](\d{1,2})[日]?',
            text
        )
        if m:
            start = f"{m.group(1)}年{m.group(2)}月{m.group(3)}日"
            end = f"{m.group(4)}年{m.group(5)}月{m.group(6)}日"
            fields["保险起期"] = start
            fields["保险止期"] = end
            fields["保险期间"] = f"{start}至{end}"
            return fields

        # 斜杠日期格式 2026/05/01
        m = re.search(
            r'(\d{4})/(\d{1,2})/(\d{1,2})\s*\d{0,2}时?\s*至\s*(\d{4})/(\d{1,2})/(\d{1,2})',
            text
        )
        if m:
            start = f"{m.group(1)}年{m.group(2)}月{m.group(3)}日"
            end = f"{m.group(4)}年{m.group(5)}月{m.group(6)}日"
            fields["保险起期"] = start
            fields["保险止期"] = end
            fields["保险期间"] = f"{start}至{end}"
            return fields

        # 短横线格式 2026-04-22
        m = re.search(
            r'保险期间[：:\s]*自\s*(\d{4}-\d{1,2}-\d{1,2})\s*起.*?至\s*(\d{4}-\d{1,2}-\d{1,2})\s*止',
            text
        )
        if m:
            start_parts = m.group(1).split("-")
            end_parts = m.group(2).split("-")
            start = f"{start_parts[0]}年{start_parts[1]}月{start_parts[2]}日"
            end = f"{end_parts[0]}年{end_parts[1]}月{end_parts[2]}日"
            fields["保险起期"] = start
            fields["保险止期"] = end
            fields["保险期间"] = f"{start}至{end}"
            return fields

        return fields

    def extract_sign_date(self, text: str, text_merged: str) -> str:
        """签单日期提取"""
        patterns = [

            r'签发[时日]间?[期]?[：:\s]*(\d{4}[年/-]\d{1,2}[月/-]\d{1,2}[日]?)',

            r'出单日期[：:\s]*(\d{4}[年/-]\d{1,2}[月/-]\d{1,2}[日]?)',
            r'制单时间[：:\s]*(\d{4}[年/-]\d{1,2}[月/-]\d{1,2}[日]?)',
        ]
        for pattern in patterns:
            m = re.search(pattern, text)
            if m:
                return m.group(1)
            m = re.search(pattern, text_merged)
            if m:
                return m.group(1)
        return ""

    def extract_fee_confirm_time(self, text: str, text_merged: str) -> str:
        m = re.search(r'收费确认时间[：:\s]*(\S+)', text)
        return m.group(1).strip() if m else ""

    def extract_sales_channel(self, text: str, text_merged: str) -> str:
        m = re.search(r'销售渠道[：:\s]*(\S+)', text)
        return m.group(1).strip() if m else ""

    def extract_intermediary(self, text: str, text_merged: str) -> str:
        patterns = [
            r'中介机构名称[：:\s]*(\S+)',
            r'经纪公司名称[：:\s]*(\S+)',
            r'保险中介名称[：:\s]*(\S+)',
        ]
        for pattern in patterns:
            m = re.search(pattern, text)
            if m:
                return m.group(1).strip()
        return ""

    def extract_dispute_resolution(self, text: str, text_merged: str) -> str:
        m = re.search(r'争议解决方式[：:\s]*(\S+)', text)
        return m.group(1).strip() if m else ""

    def extract_underwriter(self, text: str, text_merged: str) -> str:
        m = re.search(r'制单人[：:\s]*(\S+?)(?:\s|制单时间|$)', text)
        return m.group(1).strip() if m else ""

    def extract_handler_person(self, text: str, text_merged: str) -> str:
        m = re.search(r'经办人[：:\s]*(\S+)', text)
        return m.group(1).strip() if m else ""

    # ===== 险种专属/保司特有字段（子类 override）=====

    def extract_type_specific(self, text: str, text_merged: str) -> Dict[str, str]:
        """险种专属字段 — 子类 override"""

    def extract_company_specific(self, text: str, text_merged: str) -> Dict[str, str]:
        """保司特有字段 — 子类 override"""
        return {}

    def extract_items(self, text: str, text_merged: str) -> List[Dict[str, str]]:
        """险种明细列表 — 子类 override"""
        return []
```

```python
"""模板版本匹配引擎"""

import json
import os
import re
import logging
from typing import List, Dict, Optional
from insurance.engine.schema import MatchResult

logger = logging.getLogger(__name__)


class TemplateMatcher:
    """在指定保司×险种下，找到最匹配的模板版本"""

    def __init__(self, templates_dir: str):
        self.templates_dir = templates_dir
        self.registry = self._load_registry()
        self.templates_cache = {}  # id → template dict

    def _load_registry(self) -> dict:
        registry_path = os.path.join(self.templates_dir, "registry.json")
        if not os.path.exists(registry_path):
            return {"templates": []}
        with open(registry_path, "r", encoding="utf-8") as f:
            return json.load(f)

    def reload(self):
        """重新加载注册表（模板更新后调用）"""
        self.registry = self._load_registry()
        self.templates_cache.clear()

    def _load_template(self, template_entry: dict) -> Optional[dict]:
        """加载单个模板文件"""
        tid = template_entry["id"]
        if tid in self.templates_cache:
            return self.templates_cache[tid]
        filepath = os.path.join(self.templates_dir, template_entry["file"])
        if not os.path.exists(filepath):
            logger.warning(f"模板文件不存在: {filepath}")
            return None
        with open(filepath, "r", encoding="utf-8") as f:
            tpl = json.load(f)
        self.templates_cache[tid] = tpl
        return tpl

    def match(self, text: str, company_id: str, policy_type: str) -> MatchResult:
        """匹配模板版本"""
        # 筛选候选模板
        candidates = []
        for entry in self.registry.get("templates", []):
            if (entry.get("company_id") == company_id and
                entry.get("policy_type") == policy_type and
                entry.get("status") == "active"):
                tpl = self._load_template(entry)
                if tpl:
                    candidates.append(tpl)

        if not candidates:
            return MatchResult(status="no_template")

        # 计算每个候选的匹配度
        results = []
        for tpl in candidates:
            score = self._calc_score(text, tpl)
            results.append((tpl, score))
        results.sort(key=lambda x: x[1], reverse=True)
        best_tpl, best_score = results[0]

        if best_score >= 0.8:
            return MatchResult(
                template=best_tpl, template_id=best_tpl["id"],
                score=best_score, status="matched")
        elif best_score >= 0.5:
            return MatchResult(
                template=best_tpl, template_id=best_tpl["id"],
                score=best_score, status="partial_match",
                warning="模板匹配度偏低，可能存在结构变动")
        else:
            return MatchResult(
                score=best_score, status="no_match",
                warning="疑似新模板版本")

    def _calc_score(self, text: str, template: dict) -> float:
        """计算文本与模板指纹的匹配度"""
        fp = template.get("fingerprint", {})
        score = 0.0
        total_weight = 0.0

        # must_have（每个权重 0.2）
        for kw in fp.get("must_have", []):
            total_weight += 0.2
            if kw in text:
                score += 0.2

        # must_not_have（命中则扣分）
        for kw in fp.get("must_not_have", []):
            if kw in text:
                score -= 0.3

        # should_have（每个权重 0.1）
        for kw in fp.get("should_have", []):
            total_weight += 0.1
            if kw in text:
                score += 0.1

        # company_signals（每个权重 0.05）
        for signal in fp.get("company_signals", []):
            total_weight += 0.05
            if signal in text:
                score += 0.05

        # 保单号格式（权重 0.15）
        policy_no_pattern = fp.get("policy_no_pattern")
        if policy_no_pattern:
            total_weight += 0.15
            numbers = re.findall(r'[A-Za-z0-9]{10,30}', text)
            for num in numbers:
                if re.match(policy_no_pattern, num):
                    score += 0.15
                    break

        return max(0, score / total_weight) if total_weight > 0 else 0
```

```python
class FieldExtractor:
    """根据模板加载并执行对应的规则"""

    def __init__(self):
        self._rule_cache = {}

    def load_rule(self, rule_file: str) -> BaseRule:
        """加载指定的规则文件
        rule_file: 如 "picc/compulsory_v1.py"
        """
        if rule_file in self._rule_cache:
            return self._rule_cache[rule_file]

        # 将文件路径转为模块路径
        # "picc/compulsory_v1.py" → "insurance.rules.picc.compulsory_v1"
        module_path = rule_file.replace(".py", "").replace("/", ".")
        module_name = f"insurance.rules.{module_path}"

        try:
            module = importlib.import_module(module_name)
            # 找到 BaseRule 的子类
            rule_class = None
            for attr_name in dir(module):
                attr = getattr(module, attr_name)
                if (isinstance(attr, type) and issubclass(attr, BaseRule)
                        and attr is not BaseRule
                        and attr not in (BaseCompulsoryRule, BaseCommercialRule, BaseAccidentRule)):
                    rule_class = attr
                    break

            if rule_class is None:
                logger.warning(f"规则文件 {rule_file} 中未找到 BaseRule 子类")
                return BaseRule()

            rule = rule_class()
            self._rule_cache[rule_file] = rule
            return rule
        except Exception as e:
            logger.error(f"加载规则 {rule_file} 失败: {e}")
            return BaseRule()

    def load_base_rule(self, policy_type: str) -> BaseRule:
        """加载险种基础规则（无专属模板时的降级选项）"""
        rule_class = BASE_RULES.get(policy_type, BaseRule)
        return rule_class()
```

```python
class StructureValidator:
    """校验提取结果是否符合模板定义的结构"""

    def validate(self, extracted_fields: Dict, template: Dict,
                 match_score: float) -> ValidationResult:
        alerts = []
        schema = template.get("field_schema", {})

        # 1. 必填字段缺失
        missing_required = []
        for layer in ["common", "type_specific", "company_specific"]:
            for field_name, config in schema.get(layer, {}).items():
                if config.get("required"):
                    value = extracted_fields.get(layer, {}).get(field_name)
                    if not value:
                        missing_required.append(field_name)

        if missing_required:
            alerts.append(Alert(
                level="warning",
                alert_type="missing_required_fields",
                message=f"必填字段缺失: {', '.join(missing_required)}",
                template_id=template.get("id", ""),
            ))

        # 2. 模板匹配度偏低
        if match_score < 0.8:
            alerts.append(Alert(
                level="warning",
                alert_type="low_match_score",
                message=f"模板匹配度 {match_score:.0%}，可能存在结构变动",
                template_id=template.get("id", ""),
            ))

        # 3. 新字段出现（文本中有但模板未定义的）
        for layer in ["common", "type_specific", "company_specific"]:
            schema_fields = set(schema.get(layer, {}).keys())
            actual_fields = set(
                k for k, v in extracted_fields.get(layer, {}).items() if v)
            unknown = actual_fields - schema_fields
            if unknown:
                alerts.append(Alert(
                    level="info",
                    alert_type="new_fields_detected",
                    message=f"发现未定义字段({layer}): {', '.join(unknown)}",
                    template_id=template.get("id", ""),
                ))

        return ValidationResult(
            is_valid=len(missing_required) == 0,
            alerts=alerts,
        )
```

```python
import json
import logging
from typing import Dict, List, Optional

logger = logging.getLogger(__name__)


class LowMatchCollector:
    """收集低匹配度的保单，积累足够样本后触发自动训练"""

    THRESHOLD = 0.6   # 匹配度低于此值的保单会被收集
    MIN_SAMPLES = 3   # 积累到此数量后触发训练

    def __init__(self, db):
        self.db = db

    def collect(self, company_id: str, policy_type: str,
                text: str, match_score: float, record_id: int):
        """收集一条低匹配保单"""
        if match_score >= self.THRESHOLD:
            return

        from insurance.db import get_insurance_config, set_insurance_config
        key = f"low_match_samples_{company_id}_{policy_type}"
        samples = get_insurance_config(self.db, key, default=[])
        if not isinstance(samples, list):
            samples = []

        samples.append({
            "record_id": record_id,
            "text": text[:5000],  # 只存前 5000 字（节省空间）
            "match_score": match_score,
        })
        set_insurance_config(self.db, key, samples)

        logger.info(f"收集低匹配样本: {company_id}/{policy_type}，当前 {len(samples)} 份")

        # 检查是否达到触发条件
        if len(samples) >= self.MIN_SAMPLES:
            return {
                "ready": True,
                "company_id": company_id,
                "policy_type": policy_type,
                "sample_count": len(samples),
            }
        return None

    def get_samples(self, company_id: str, policy_type: str) -> List[str]:
        """获取收集的样本文本"""
        from insurance.db import get_insurance_config
        key = f"low_match_samples_{company_id}_{policy_type}"
        samples = get_insurance_config(self.db, key, default=[])
        return [s["text"] for s in samples if isinstance(s, dict)]

    def clear_samples(self, company_id: str, policy_type: str):
        """清除收集的样本（训练完成后调用）"""
        from insurance.db import set_insurance_config
        key = f"low_match_samples_{company_id}_{policy_type}"
        set_insurance_config(self.db, key, [])
```

```python
import os
from typing import List, Dict, Optional
from insurance.engine.schema import (
    PipelineResult, CompanyResult, PolicyTypeResult,
    Alert, ValidationResult, FieldDiff,

# 文档分类关键词
DOC_CATEGORY_RULES = [
    # (条件函数, 分类名)
    (lambda t, f: len(t) < 500 and "保险人签章" in t, "电子标志"),
    (lambda t, f: "此标志粘贴在机动车前窗" in t, "电子标志"),
    (lambda t, f: "此标志正面的年份" in t, "电子标志"),

    (lambda t, f: "缴款通知书" in t[:500], "缴款通知"),
    (lambda t, f: "免责事项说明书" in t or "投保人声明" in t, "条款"),
    (lambda t, f: "保险示范条款" in t, "条款"),
    (lambda t, f: "(cid:" in t[:200], "其他"),
]


class InsurancePipeline:
    """保单识别主流水线"""

    def __init__(self, db=None, ocr_engines=None, alert_webhook=""):
        templates_dir = os.path.join(
            os.path.dirname(os.path.dirname(__file__)), "templates")

        self.company_identifier = CompanyIdentifier()
        self.type_identifier = TypeIdentifier()
        self.template_matcher = TemplateMatcher(templates_dir)
        self.field_extractor = FieldExtractor()
        self.structure_validator = StructureValidator()
        self.ocr_dispatcher = OCRDispatcher(engines=ocr_engines or [])
        self.ocr_comparator = OCRComparator()
        self.alert_notifier = AlertNotifier(db, alert_webhook) if db else None
        self.low_match_collector = LowMatchCollector(db) if db else None

    def process_text(self, text: str, filename: str = "") -> PipelineResult:
        """直接处理文本（已有文本的情况）"""
        return self._process(text, filename=filename, ocr_engines=["text_input"])

    def process_pdf(self, pdf_bytes: bytes, filename: str = "") -> PipelineResult:
        """处理 PDF 文件"""
        # OCR 阶段
        ocr_results = self.ocr_dispatcher.dispatch(pdf_bytes)

        # 获取主文本
        success_results = [r for r in ocr_results if r.success]
        if not success_results:
            return PipelineResult(
                confidence=0,
                doc_category="其他",
                ocr_engines=[r.engine for r in ocr_results],
            )

        primary_text = success_results[0].text
        ocr_engine_names = [r.engine for r in success_results]
        is_scan = len(success_results) > 1  # 多引擎 = 扫描件

        # 走主流程
        result = self._process(
            primary_text, filename=filename,
            ocr_results=success_results if is_scan else None,
            ocr_engines=ocr_engine_names,
        )
        return result

    def _process(self, text: str, filename: str = "",
                 ocr_results: List[OCRResult] = None,
                 ocr_engines: List[str] = None) -> PipelineResult:
        """主流程"""
        # Stage 1: 识别保司
        company = self.company_identifier.identify(text)

        # Stage 2: 识别险种
        policy_type = self.type_identifier.identify(text)

        # Stage 3: 匹配模板版本
        match = self.template_matcher.match(
            text, company.company_id, policy_type.type_code)

        # 加载规则
        if match.template:
            rule = self.field_extractor.load_rule(match.template["rule_file"])
        else:
            rule = self.field_extractor.load_base_rule(policy_type.type_code)

        # Stage 4: 字段提取
        ocr_diffs = []
        if ocr_results and len(ocr_results) > 1:
            # 扫描件：多 OCR 交叉比对
            compare_report = self.ocr_comparator.compare(
                ocr_results, lambda t: rule.extract_all(t))
            extracted = compare_report.merged_fields
            ocr_diffs = compare_report.diffs
        else:
            # 文本层 / 单 OCR：直接提取
            extracted = rule.extract_all(text)

        # Stage 5: 结构校验
        validation = ValidationResult()
        if match.template:
            validation = self.structure_validator.validate(
                extracted, match.template, match.score)

        # 告警通知
        if self.alert_notifier:
            context = {
                "company_short": company.company_short,
                "policy_type": policy_type.type_name,
                "filename": filename,
                "match_score": match.score,
            }
            for alert in validation.alerts:
                self.alert_notifier.notify(alert, context)
            # OCR 关键字段差异告警
            critical_diffs = [d for d in ocr_diffs if d.is_critical]
            if critical_diffs:
                diff_detail = "; ".join(
                    f"{d.field_name}: {d.values}" for d in critical_diffs)
                self.alert_notifier.notify(
                    Alert(level="warning", alert_type="ocr_critical_diff",
                          message=f"OCR 关键字段差异: {diff_detail}"),
                    context)

        # 文档分类
        doc_category = self._identify_doc_category(text, extracted)

        # 置信度
        key_fields = ["保单号", "被保险人", "车牌号", "保险起期", "保费合计"]
        common = extracted.get("common", {})
        hit_count = sum(1 for f in key_fields if common.get(f))
        confidence = hit_count / len(key_fields)

        # Stage 6: 字段映射
        # 兼容旧版 apply_mapping：将三层字段展平
        flat_fields = {}
        flat_fields.update(extracted.get("common", {}))
        flat_fields.update(extracted.get("type_specific", {}))
        flat_fields.update(extracted.get("company_specific", {}))
        flat_fields["保险公司"] = company.company_name
        flat_fields["保险公司简称"] = company.company_short
        flat_fields["险种类型"] = policy_type.type_name
        mapped = apply_mapping(flat_fields, company.company_short)

        return PipelineResult(
            company=company,
            policy_type=policy_type,
            template_id=match.template_id,
            match_score=match.score,
            fields=extracted,
            mapped_fields=mapped,
            validation=validation,
            ocr_diffs=ocr_diffs,
            doc_category=doc_category,
            confidence=confidence,
            ocr_engines=ocr_engines or [],
            insurance_items=extracted.get("insurance_items", []),
            raw_text=text,
        )

    def _identify_doc_category(self, text: str, fields: Dict) -> str:

        for check_fn, category in DOC_CATEGORY_RULES:
            try:
                if check_fn(text, fields):
                    return category
            except Exception:
                continue
        return "保单"
```

```python
"""多文本结构对齐 — 找出固定部分和变量部分"""

import os
import re
from dataclasses import dataclass, field
from typing import List, Optional


@dataclass
class AlignedLine:
    """对齐后的行"""
    index: int = 0
    line_type: str = ""      # fixed / label_value / variable
    content: str = ""        # 固定内容或公共前缀
    label: str = ""          # 标签（label_value 类型）
    variants: List[str] = field(default_factory=list)  # 各样本的值


@dataclass
class AlignmentResult:
    """对齐结果"""
    lines: List[AlignedLine] = field(default_factory=list)
    sample_count: int = 0
    fixed_lines: int = 0
    variable_lines: int = 0
    label_value_lines: int = 0


class TextAligner:
    """将 N 份同类保单文本按行对齐"""

    def align(self, texts: List[str]) -> AlignmentResult:
        if len(texts) < 2:
            return AlignmentResult(sample_count=len(texts))

        line_groups = [text.split('\n') for text in texts]
        max_lines = max(len(g) for g in line_groups)

        aligned_lines = []
        fixed_count = 0
        variable_count = 0
        label_value_count = 0

        for line_idx in range(max_lines):
            lines_at_idx = []
            for group in line_groups:
                if line_idx < len(group):
                    lines_at_idx.append(group[line_idx].strip())
                else:
                    lines_at_idx.append("")

            non_empty = [l for l in lines_at_idx if l]
            if not non_empty:
                continue

            unique_lines = set(non_empty)

            if len(unique_lines) == 1:
                # 所有样本该行完全一致 → 固定文本
                aligned_lines.append(AlignedLine(
                    index=line_idx, line_type="fixed",
                    content=non_empty[0]))
                fixed_count += 1
            else:
                # 有差异 → 分析
                prefix, suffix = self._find_common_affixes(non_empty)
                if prefix and len(prefix) >= 2:
                    # 有公共前缀 → 标签：值 模式
                    label = prefix.rstrip('：: \t')
                    values = []
                    for line in lines_at_idx:
                        if line and prefix:
                            val = line[len(prefix):].strip()
                            if suffix and val.endswith(suffix):
                                val = val[:-len(suffix)].strip()
                            values.append(val)
                        else:
                            values.append(line)
                    aligned_lines.append(AlignedLine(
                        index=line_idx, line_type="label_value",
                        content=prefix, label=label,
                        variants=values))
                    label_value_count += 1
                else:
                    # 整行不同 → 纯变量行
                    aligned_lines.append(AlignedLine(
                        index=line_idx, line_type="variable",
                        variants=list(unique_lines)))
                    variable_count += 1

        return AlignmentResult(
            lines=aligned_lines,
            sample_count=len(texts),
            fixed_lines=fixed_count,
            variable_lines=variable_count,
            label_value_lines=label_value_count,
        )

    def _find_common_affixes(self, lines: List[str]) -> tuple:
        """找多行的公共前缀和后缀"""
        if not lines:
            return "", ""
        prefix = os.path.commonprefix(lines)
        reversed_lines = [l[::-1] for l in lines]
        suffix = os.path.commonprefix(reversed_lines)[::-1]
        # 避免前后缀重叠
        if len(prefix) + len(suffix) >= min(len(l) for l in lines):
            suffix = ""
        return prefix, suffix
```

```python
"""正则自动归纳 — 从样本值推导提取正则"""

import re
from dataclasses import dataclass
from typing import List
from insurance.learner.field_discoverer import DiscoveredField


@dataclass
class InducedPattern:
    """归纳出的正则模式"""
    field_name: str = ""
    label_pattern: str = ""     # 标签的正则
    value_pattern: str = ""     # 值的正则
    full_pattern: str = ""      # 完整提取正则（标签+分隔符+值）
    source: str = ""            # preset / induced
    confidence: float = 0.0


# 字段预定义模式
FIELD_PRESETS = {
    "车牌号": r'[京津沪渝冀豫云辽黑湘皖鲁新苏浙赣鄂桂甘晋蒙陕吉闽贵粤川青藏琼宁][A-Z][A-Z0-9]{5,6}',
    "保费合计": r'[\d,]+\.?\d*',
    "不含税保费": r'[\d,]+\.?\d*',
    "增值税额": r'[\d,]+\.?\d*',
    "车船税": r'[\d,]+\.?\d*',
    "车架号VIN": r'[A-HJ-NPR-Z0-9]{17}',
}


class PatternInducer:
    """从样本值自动归纳正则"""

    def induce(self, field: DiscoveredField) -> InducedPattern:
        values = [v for v in field.sample_values if v and v.strip()]

        # 构建标签正则（字符间允许空格）
        label_pattern = self._build_label_regex(field.original_label)
        separator = r'[：:\s]*\s*'

        # 预定义模式
        if field.standard_name in FIELD_PRESETS:
            preset = FIELD_PRESETS[field.standard_name]
            full = f'{label_pattern}{separator}({preset})'
            return InducedPattern(
                field_name=field.standard_name,
                label_pattern=label_pattern,
                value_pattern=preset,
                full_pattern=full,
                source="preset",
                confidence=0.95,
            )

        # 自动分析
        value_pattern = self._analyze_values(values)
        full = f'{label_pattern}{separator}({value_pattern})'

        return InducedPattern(
            field_name=field.standard_name,
            label_pattern=label_pattern,
            value_pattern=value_pattern,
            full_pattern=full,
            source="induced",
            confidence=self._calc_confidence(values, value_pattern),
        )

    def _analyze_values(self, values: List[str]) -> str:
        """分析值类型，归纳正则"""
        if not values:
            return r'\S+'

        # 纯数字
        if all(re.match(r'^[\d,.]+$', v) for v in values):
            return r'[\d,]+\.?\d*'

        # 纯中文
        if all(re.match(r'^[\u4e00-\u9fa5]+$', v) for v in values):
            lengths = [len(v) for v in values]
            return rf'[\u4e00-\u9fa5]{{{max(1, min(lengths) - 1)},{max(lengths) + 2}}}'

        # 纯字母数字
        if all(re.match(r'^[A-Za-z0-9]+$', v) for v in values):
            lengths = [len(v) for v in values]
            return rf'[A-Za-z0-9]{{{max(1, min(lengths) - 1)},{max(lengths) + 2}}}'

        # 日期格式
        if all(re.match(r'^\d{4}[年/-]\d{1,2}[月/-]\d{1,2}', v) for v in values):
            return r'\d{4}[年/-]\d{1,2}[月/-]\d{1,2}[日]?'

        # 混合
        return r'\S+'

    def _build_label_regex(self, label: str) -> str:
        """标签 → 容错正则"""
        label = label.strip()
        if not label:
            return ""
        chars = list(label)
        return r'\s*'.join(re.escape(c) for c in chars)

    def _calc_confidence(self, values: List[str], pattern: str) -> float:
        """计算正则对样本值的匹配率"""
        if not values:
            return 0
        match_count = sum(1 for v in values if re.match(pattern, v))
        return match_count / len(values)
```

```python
"""自验证 — 用生成的规则反跑所有样本"""

import re
from dataclasses import dataclass, field
from typing import List, Dict
from insurance.learner.field_discoverer import DiscoveredField, FIELD_LAYER
from insurance.rules.base_rule import BaseRule, merge_text


@dataclass
class FieldValidation:
    field_name: str = ""
    expected: str = ""
    actual: str = ""
    match: bool = False


@dataclass
class SampleValidation:
    sample_index: int = 0
    fields: List[FieldValidation] = field(default_factory=list)
    hit_rate: float = 0.0


@dataclass
class ValidationReport:
    sample_count: int = 0
    overall_hit_rate: float = 0.0
    passed: bool = False
    unstable_fields: List[Dict] = field(default_factory=list)
    details: List[SampleValidation] = field(default_factory=list)


class SelfValidator:
    """用生成的规则反跑所有样本，验证准确率"""

    def validate(self, rule: BaseRule, samples: List[str],
                 discovered_fields: List[DiscoveredField]) -> ValidationReport:

        results = []
        for i, text in enumerate(samples):
            extracted = rule.extract_all(text)
            field_results = []

            for df in discovered_fields:
                expected = df.sample_values[i] if i < len(df.sample_values) else ""
                if not expected:
                    continue

                layer = FIELD_LAYER.get(df.standard_name, "company_specific")
                actual = extracted.get(layer, {}).get(df.standard_name, "")

                matched = self._fuzzy_match(expected, str(actual) if actual else "")
                field_results.append(FieldValidation(
                    field_name=df.standard_name,
                    expected=expected,
                    actual=str(actual) if actual else "",
                    match=matched,
                ))

            hit_rate = (sum(1 for f in field_results if f.match) / len(field_results)
                       if field_results else 0)
            results.append(SampleValidation(
                sample_index=i, fields=field_results, hit_rate=hit_rate))

        overall = sum(r.hit_rate for r in results) / len(results) if results else 0

        # 找不稳定字段
        unstable = []
        for df in discovered_fields:
            matches = [fr for sr in results for fr in sr.fields
                      if fr.field_name == df.standard_name]
            fail_count = sum(1 for m in matches if not m.match)
            if fail_count > 0:
                unstable.append({
                    "field": df.standard_name,
                    "fail_count": fail_count,
                    "total": len(matches),
                })

        return ValidationReport(
            sample_count=len(samples),
            overall_hit_rate=overall,
            passed=overall >= 0.9 and len(unstable) == 0,
            unstable_fields=unstable,
            details=results,
        )

    def _fuzzy_match(self, expected: str, actual: str) -> bool:
        """模糊匹配（忽略空格、标点差异）"""
        if not expected and not actual:
            return True
        if not expected or not actual:
            return False
        # 去除空格和常见标点后比较
        clean_exp = re.sub(r'[\s,，、。：:；;]', '', expected)
        clean_act = re.sub(r'[\s,，、。：:；;]', '', actual)
        if clean_exp == clean_act:
            return True
        # 包含关系
        if clean_exp in clean_act or clean_act in clean_exp:
            return True
        return False
```

```python
class TrainResult:
    def __init__(self, success=False, error="", template=None, rule_code="",
                 validation=None, needs_review=None):
        self.success = success
        self.error = error
        self.template = template          # 模板 JSON dict
        self.rule_code = rule_code        # 规则 Python 代码
        self.validation = validation      # ValidationReport
        self.needs_review = needs_review or []


class TemplateTrainer:
    """主入口：接收样本 PDF，输出模板 + 规则"""

    def __init__(self):
        self.ocr = PdfplumberEngine()
        self.aligner = TextAligner()
        self.discoverer = FieldDiscoverer()
        self.inducer = PatternInducer()
        self.rule_gen = RuleGenerator()
        self.template_gen = TemplateGenerator()
        self.validator = SelfValidator()
        self.company_id_helper = CompanyIdentifier()

    def train(self, pdf_bytes_list: List[bytes],
              company_id: str, policy_type: str,
              version: str = "v1",
              company_name: str = "",
              company_short: str = "") -> TrainResult:
        """
        训练模板

        Args:
            pdf_bytes_list: N 份同类保单 PDF
            company_id: 保司 ID（如 "picc"）
            policy_type: 险种（compulsory / commercial / accident）
            version: 版本号
            company_name: 保司全称（可选，会自动识别）
            company_short: 保司简称（可选）
        """
        # Step 1: 提取文本
        texts = []
        for pdf_bytes in pdf_bytes_list:
            result = self.ocr.extract(pdf_bytes)
            if result.success:
                texts.append(result.text)
        if len(texts) < 2:
            return TrainResult(
                success=False, error=f"有效样本不足，需要至少 2 份（当前 {len(texts)} 份）")

        # 自动识别保司信息
        if not company_name or not company_short:
            company_result = self.company_id_helper.identify(texts[0])
            if not company_name:
                company_name = company_result.company_name
            if not company_short:
                company_short = company_result.company_short

        # Step 2: 结构对齐
        alignment = self.aligner.align(texts)
        logger.info(f"对齐完成: 固定行 {alignment.fixed_lines}, "
                    f"标签值行 {alignment.label_value_lines}, "
                    f"变量行 {alignment.variable_lines}")

        # Step 3: 字段发现
        fields = self.discoverer.discover(alignment)
        logger.info(f"发现 {len(fields)} 个字段")

        if not fields:
            return TrainResult(
                success=False, error="未能自动发现任何字段，样本差异过大或格式不标准")

        # Step 4: 正则归纳
        patterns = []
        for field in fields:
            pattern = self.inducer.induce(field)
            patterns.append(pattern)

        # Step 5: 生成模板 + 规则
        template = self.template_gen.generate(
            company_id, company_short, company_name,
            policy_type, version, fields, patterns, alignment)
        rule_code = self.rule_gen.generate(
            company_id, policy_type, version,
            fields, patterns, sample_count=len(texts))

        # Step 6: 自验证
        rule_instance = self._load_rule_from_code(rule_code, policy_type)
        validation = self.validator.validate(rule_instance, texts, fields)

        logger.info(f"训练完成: 准确率 {validation.overall_hit_rate:.1%}, "
                    f"通过={validation.passed}, "
                    f"不稳定字段={len(validation.unstable_fields)}")

        return TrainResult(
            success=validation.passed,
            template=template,
            rule_code=rule_code,
            validation=validation,
            needs_review=validation.unstable_fields,
        )

    def train_from_texts(self, texts: List[str],
                         company_id: str, policy_type: str,
                         version: str = "v1",
                         company_name: str = "",
                         company_short: str = "") -> TrainResult:
        """从已提取的文本训练（跳过 OCR 步骤）"""
        if len(texts) < 2:
            return TrainResult(success=False, error="至少需要 2 份样本")

        if not company_name or not company_short:
            company_result = self.company_id_helper.identify(texts[0])
            if not company_name:
                company_name = company_result.company_name
            if not company_short:
                company_short = company_result.company_short

        alignment = self.aligner.align(texts)
        fields = self.discoverer.discover(alignment)
        if not fields:
            return TrainResult(success=False, error="未能发现字段")

        patterns = [self.inducer.induce(f) for f in fields]
        template = self.template_gen.generate(
            company_id, company_short, company_name,
            policy_type, version, fields, patterns, alignment)
        rule_code = self.rule_gen.generate(
            company_id, policy_type, version, fields, patterns, len(texts))

        rule_instance = self._load_rule_from_code(rule_code, policy_type)
        validation = self.validator.validate(rule_instance, texts, fields)

        return TrainResult(
            success=validation.passed,
            template=template,
            rule_code=rule_code,
            validation=validation,
            needs_review=validation.unstable_fields,
        )

    def _load_rule_from_code(self, code: str, policy_type: str) -> BaseRule:
        """动态加载生成的规则代码"""
        # 创建临时模块
        module = types.ModuleType("_temp_rule")
        # 注入依赖
        import insurance.rules.base_rule
        import insurance.rules.base_compulsory
        import insurance.rules.base_commercial
        import insurance.rules.base_accident
        module.__dict__["re"] = __import__("re")

        base_map = {
            "compulsory": ("BaseCompulsoryRule", insurance.rules.base_compulsory),
            "commercial": ("BaseCommercialRule", insurance.rules.base_commercial),
            "accident": ("BaseAccidentRule", insurance.rules.base_accident),
        }
        if policy_type in base_map:
            cls_name, mod = base_map[policy_type]
            module.__dict__[cls_name] = getattr(mod, cls_name)
        else:
            module.__dict__["BaseRule"] = BaseRule

        try:
            exec(code, module.__dict__)
            # 找到生成的规则类
            for name in dir(module):
                obj = getattr(module, name)
                if (isinstance(obj, type) and issubclass(obj, BaseRule)
                        and obj not in (BaseRule, BaseCompulsoryRule,
                                       BaseCommercialRule, BaseAccidentRule)):
                    return obj()
        except Exception as e:
            logger.error(f"动态加载规则失败: {e}")

        # 降级返回基础规则
        base_classes = {
            "compulsory": BaseCompulsoryRule,
            "commercial": BaseCommercialRule,
            "accident": BaseAccidentRule,
        }
        return base_classes.get(policy_type, BaseRule)()


def save_training_result(templates_dir: str, rules_dir: str,
                         result: TrainResult, company_id: str,
                         policy_type: str, version: str):
    """保存训练结果到文件系统"""
    if not result.success and not result.template:
        return

    # 保存模板 JSON
    tpl_dir = os.path.join(templates_dir, company_id, policy_type)
    os.makedirs(tpl_dir, exist_ok=True)
    tpl_path = os.path.join(tpl_dir, f"{version}.json")
    with open(tpl_path, "w", encoding="utf-8") as f:
        json.dump(result.template, f, ensure_ascii=False, indent=2)

    # 保存规则 .py
    rule_dir = os.path.join(rules_dir, company_id)
    os.makedirs(rule_dir, exist_ok=True)
    # 写入 __init__.py
    init_path = os.path.join(rule_dir, "__init__.py")
    if not os.path.exists(init_path):
        with open(init_path, "w") as f:
            pass
    rule_path = os.path.join(rule_dir, f"{policy_type}_{version}.py")
    with open(rule_path, "w", encoding="utf-8") as f:
        f.write(result.rule_code)

    # 更新注册表
    registry_path = os.path.join(templates_dir, "registry.json")
    with open(registry_path, "r", encoding="utf-8") as f:
        registry = json.load(f)

    template_id = f"{company_id}/{policy_type}/{version}"
    # 去重
    registry["templates"] = [
        t for t in registry["templates"] if t["id"] != template_id]
    registry["templates"].append({
        "id": template_id,
        "company_id": company_id,
        "policy_type": policy_type,
        "version": version,
        "status": "active",
        "file": f"{company_id}/{policy_type}/{version}.json",
    })

    with open(registry_path, "w", encoding="utf-8") as f:
        json.dump(registry, f, ensure_ascii=False, indent=2)

    logger.info(f"训练结果已保存: 模板 {tpl_path}, 规则 {rule_path}")
```
